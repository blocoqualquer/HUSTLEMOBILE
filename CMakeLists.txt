cmake_minimum_required(VERSION 3.5)
if(NOT DEFINED ENV{PSPDEV})
  message(FATAL_ERROR "defina PSPDEV (ex.: export PSPDEV=/usr/local/pspdev) e use psp-cmake")
endif()
project(yomihustle_psp C)
set(CMAKE_C_STANDARD 99)
add_executable(yomihustle src/main.c src/gfx.c)
target_compile_options(yomihustle PRIVATE -Wall -Wextra -O2 -G0)
target_link_libraries(yomihustle pspgu pspdisplay pspge pspctrl pspaudio m)
create_pbp_file(TARGET yomihustle TITLE "YOMI HUSTLE PSP")
